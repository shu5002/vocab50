    function pick(arr) {

      return arr[Math.floor(Math.random() * arr.length)];

    }


    function cap(str) {

      return str.charAt(0).toUpperCase() + str.slice(1);

    }


    function shuffleChoices(correctText, distractors) {

      const uniqueWrong = Array.from(new Set(distractors.filter(d => d !== correctText))).slice(0, 3);

      const combined = [correctText, ...uniqueWrong];

      for (let i = combined.length - 1; i > 0; i--) {

        const j = Math.floor(Math.random() * (i + 1));

        [combined[i], combined[j]] = [combined[j], combined[i]];

      }

      return {

        choices: combined,

        ans: combined.indexOf(correctText)

      };

    }

          return {

          };

    let audioCtxUnlocked = false;

    function unlockAudio() {

      if (audioCtxUnlocked) return;

      try {

        const AC = window.AudioContext || window.webkitAudioContext;

        if (AC) {

          const ctx = new AC();

          const buf = ctx.createBuffer(1, 1, 22050);

          const src = ctx.createBufferSource();

          src.buffer = buf;

          src.connect(ctx.destination);

          src.start(0);

        }

        audioCtxUnlocked = true;

      } catch (e) {}

    }

    document.addEventListener("touchstart", unlockAudio, { once: true });

    document.addEventListener("click", unlockAudio, { once: true });


    function speakEnglishSentence(text) {

      if (!text || !("speechSynthesis" in window)) return;

      const btn = document.getElementById("btn-q-speak");

      if (btn) btn.classList.add("speaking-active");

      const done = () => { if (btn) btn.classList.remove("speaking-active"); };


      try {

        window.speechSynthesis.cancel();

        const utter = new SpeechSynthesisUtterance(text);

        utter.lang = state.accent || "en-US";

        utter.rate = Number(state.speed) || 1.0;


        const voices = window.speechSynthesis.getVoices();

        if (voices && voices.length > 0) {

          const pref = voices.find(v =>

            v.lang.replace("_", "-") === utter.lang &&

            (v.name.includes("Samantha") || v.name.includes("Daniel") || v.name.includes("Google") || v.localService)

          ) || voices.find(v => v.lang.startsWith("en"));

          if (pref) utter.voice = pref;

        }

        utter.onend = done;

        utter.onerror = done;

        window.speechSynthesis.speak(utter);

        setTimeout(done, 4000);

      } catch (e) {

        done();

      }

    }


    function getCompleteSentence(q) {

      const answerWord = q.choices[q.ans];

      return q.sentence.replace(/\(\s*\)/, answerWord);

    }


    function speakCurrentSentence() {

      const q = state.currentQ;

      if (!q) return;

      const text = state.answeredCurrent

        ? getCompleteSentence(q)

        : q.sentence.replace(/\(\s*\)/, " ... ");

      speakEnglishSentence(text);

    }

    function loadNextQuestion() {

      if (state.selectedCategory === "WEAK" && state.weakQueue.length > 0) {

        const idx = Math.floor(Math.random() * state.weakQueue.length);

        state.currentQ = { ...state.weakQueue[idx], isFromWeak: true };

      } else {

        if (state.selectedCategory === "WEAK") {

          state.selectedCategory = "ALL";

        }

        state.currentQ = generateQuestion(state.selectedCategory);

      }

      state.answeredCurrent = false;

      state.lastChoiceIdx = null;

      saveState();

      renderAll();

    }


    function handleAnswer(choiceIdx) {

      if (state.answeredCurrent) return;

      const q = state.currentQ;

      const isCorrect = choiceIdx === q.ans;


      state.answeredCurrent = true;

      state.lastChoiceIdx = choiceIdx;

      state.totalAttempts = (state.totalAttempts || 0) + 1;


      if (!state.catStats[q.cat]) state.catStats[q.cat] = { attempts: 0, correct: 0 };

      state.catStats[q.cat].attempts++;


      if (isCorrect) {

        state.totalCorrect = (state.totalCorrect || 0) + 1;

        state.catStats[q.cat].correct++;

        // 弱点キューに同じ文があれば卒業（削除）

        state.weakQueue = state.weakQueue.filter(item => item.sentence !== q.sentence);

      } else {

        // 間違えた問題を弱点キューに保存

        if (!state.weakQueue.some(item => item.sentence === q.sentence)) {

          state.weakQueue.push(q);

        }

      }


      saveState();

      renderAll();


      if (state.autoSpeak) {

        speakEnglishSentence(getCompleteSentence(q));

      }

    }

    function saveSettings() {

      state.autoSpeak = document.getElementById("setting-auto-speak").checked;

      state.accent = document.getElementById("setting-accent").value;

      state.speed = parseFloat(document.getElementById("setting-speed").value) || 1.0;

      saveState();

    }


    function loadState() {

      try {

        const raw = localStorage.getItem(STORAGE_KEY) || localStorage.getItem("GRAMMAR_SPRINT_1Q_V1");

        if (raw) {

          state = { ...state, ...JSON.parse(raw) };

        }

      } catch (e) {}

      if (!Array.isArray(state.weakQueue)) state.weakQueue = [];

      if (!state.catStats) state.catStats = {};

      if (!state.currentQ) {

        state.currentQ = generateQuestion("ALL");

      }

    }


    function saveState() {

      try {

        localStorage.setItem(STORAGE_KEY, JSON.stringify(state));

      } catch (e) {}

    }


    function exportBackup() {

      const blob = new Blob([JSON.stringify(state, null, 2)], { type: "application/json" });

      const url = URL.createObjectURL(blob);

      const a = document.createElement("a");

      a.href = url;

      a.download = `grammar_infinite_backup_${new Date().toISOString().slice(0, 10)}.json`;

      a.click();

    }


    function importBackup(e) {

      const file = e.target.files?.[0];

      if (!file) return;

      const reader = new FileReader();

      reader.onload = () => {

        try {

          state = { ...state, ...JSON.parse(reader.result) };

          saveState();

          renderAll();

          alert("✅ バックアップを復元しました！");

        } catch (err) {

          alert("JSONファイルが正しくありません。");

        }

      };

      reader.readAsText(file);

    }


    window.addEventListener("DOMContentLoaded", () => {

      loadState();

      renderAll();

    });
